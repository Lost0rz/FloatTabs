Can you write a Python script to fetch the latest space news from an API?
Certainly. Here is a simple Python script using the `requests` library to fetch the latest space news from the Spaceflight News API.
python content_copy 
import requests
import json

def fetch_space_news():
    url = "https://api.spaceflightnewsapi.net/v4/articles/?limit=3"
    try:
        response = requests.get(url)
        response.raise_for_status()
        data = response.json()
        
        for article in data['results']:
            print(f"Title: {article['title']}")
            print(f"URL: {article['url']}\n")
            
    except requests.exceptions.RequestException as e:
        print(f"Error fetching data: {e}")

if __name__ == "__main__":
    fetch_space_news()
